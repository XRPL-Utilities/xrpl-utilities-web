Python, pay in XRP
pip install x402-xrpl

from x402_xrpl import X402RequestsSession
from xrpl.wallet import Wallet

def pay_in_xrp(accepts, *_):
    return next(a for a in accepts
                if a["network"] == "xrpl:0" and a["asset"] == "XRP")

session = X402RequestsSession(
    wallet=Wallet.from_seed(seed),
    rpc_url="https://YOUR_XRPL_RPC",
    payment_requirements_selector=pay_in_xrp,
)

resp = session.post(
    "https://sentinel.xrpl-utilities.io/scan",
    json={"address": "rExample..."},
)
resp.raise_for_status()  # a 402 here means the payment did not go through
r = resp.json()
Node.js, pay in USDC on Base
npm install @x402/fetch @x402/evm viem

import { wrapFetchWithPaymentFromConfig } from "@x402/fetch";
import { ExactEvmScheme } from "@x402/evm";
import { privateKeyToAccount } from "viem/accounts";

const account = privateKeyToAccount(process.env.BASE_PRIVATE_KEY);
const fetchWithPayment = wrapFetchWithPaymentFromConfig(fetch, {
  schemes: [{ network: "eip155:8453", client: new ExactEvmScheme(account) }],
});

const res = await fetchWithPayment("https://sentinel.xrpl-utilities.io/scan", {
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({ address: "rExample..." }),
});
if (res.status === 402) throw new Error("payment did not go through");
console.log(await res.json());
